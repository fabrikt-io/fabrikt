package {{package}}

import feign.RequestTemplate
import feign.codec.EncodeException
import feign.codec.Encoder
import java.io.ByteArrayOutputStream
import java.lang.reflect.Type
import java.util.UUID

@Target(AnnotationTarget.FUNCTION)
@Retention(AnnotationRetention.RUNTIME)
annotation class FeignJsonParts(vararg val names: String)

data class FeignMultipartFile(
    val bytes: ByteArray,
    val filename: String,
    val contentType: String = "application/octet-stream",
)

/** Register on Feign's builder; delegates requests without generated multipart metadata. */
class FeignMultipartEncoder(
    private val delegate: Encoder,
    private val serializeJson: (Any) -> ByteArray,
) : Encoder {
    override fun encode(value: Any?, bodyType: Type, template: RequestTemplate) {
        val jsonParts = template.methodMetadata()?.method()?.getAnnotation(FeignJsonParts::class.java)
        if (jsonParts == null) {
            delegate.encode(value, bodyType, template)
            return
        }
        val parts = value as? Map<*, *> ?: throw EncodeException("Expected named multipart parts")
        val boundary = "fabrikt-" + UUID.randomUUID()
        val output = ByteArrayOutputStream()
        fun text(value: String) { output.write(value.toByteArray(Charsets.UTF_8)) }
        fun quoted(value: String): String = value.replace("\r", "%0D").replace("\n", "%0A").replace("\"", "%22")
        fun part(name: String, bytes: ByteArray, contentType: String, filename: String? = null) {
            require(contentType.none { it == '\r' || it == '\n' }) { "Invalid multipart content type" }
            text("--$boundary\r\nContent-Disposition: form-data; name=\"${quoted(name)}\"")
            if (filename != null) text("; filename=\"${quoted(filename)}\"")
            text("\r\nContent-Type: $contentType\r\n\r\n")
            output.write(bytes)
            text("\r\n")
        }
        parts.forEach { (key, item) ->
            val name = key as? String ?: throw EncodeException("Expected a string multipart name")
            when {
                item == null -> Unit
                name in jsonParts.names -> part(name, serializeJson(item), "application/json")
                item is FeignMultipartFile -> part(name, item.bytes, item.contentType, item.filename)
                item is Iterable<*> -> item.forEach { file ->
                    require(file is FeignMultipartFile) { "Expected binary multipart files" }
                    part(name, file.bytes, file.contentType, file.filename)
                }
                else -> part(name, item.toString().toByteArray(Charsets.UTF_8), "text/plain; charset=UTF-8")
            }
        }
        text("--$boundary--\r\n")
        template.removeHeader("Content-Type")
        template.header("Content-Type", "multipart/form-data; boundary=$boundary")
        template.body(output.toByteArray(), null)
    }
}
